1. 기본 개념

그래프를 C언어에서 행렬로 저장할 때는 보통 2차원 배열을 사용합니다.

graph[a][b] = 1; 이면 a번 정점과 b번 정점이 연결되어 있다는 뜻입니다.

4.1 간선 목록으로 무방향 그래프 입력

#include <stdio.h>

int main() {
    int n, m;
    int graph[101][101] = {0};

    scanf("%d %d", &n, &m);

    for (int i = 0; i < m; i++) {
        int a, b;
        scanf("%d %d", &a, &b);
        graph[a][b] = 1;
        graph[b][a] = 1;
    }

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) printf("%d ", graph[i][j]);
        printf("\n");
    }
    return 0;
}

4.5 문자 그래프를 행렬로 입력받기

#include <stdio.h>

int main() {
    int n;
    int graph[101][101];
    char line[101];

    scanf("%d", &n);

    for (int i = 1; i <= n; i++) {
        scanf("%s", line);
        for (int j = 1; j <= n; j++) {
            graph[i][j] = line[j - 1] - '0';
        }
    }

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) printf("%d ", graph[i][j]);
        printf("\n");
    }
    return 0;
}

4.6 도시 이름을 번호로 바꾸어 입력받기

#include <stdio.h>

int getIndex(char c) {
    return c - 'A' + 1;
}

int main() {
    int n, m;
    int graph[101][101] = {0};

    scanf("%d %d", &n, &m);

    for (int i = 0; i < m; i++) {
        char a, b;
        scanf(" %c %c", &a, &b);
        int x = getIndex(a);
        int y = getIndex(b);
        graph[x][y] = 1;
        graph[y][x] = 1;
    }

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) printf("%d ", graph[i][j]);
        printf("\n");
    }
    return 0;
}

4.7 연결 여부를 빠르게 확인하기

#include <stdio.h>

int main() {
    int n, m, q;
    int graph[101][101] = {0};

    scanf("%d %d", &n, &m);

    for (int i = 0; i < m; i++) {
        int a, b;
        scanf("%d %d", &a, &b);
        graph[a][b] = 1;
        graph[b][a] = 1;
    }

    scanf("%d", &q);

    for (int i = 0; i < q; i++) {
        int a, b;
        scanf("%d %d", &a, &b);
        if (graph[a][b] == 1) printf("YES\n");
        else printf("NO\n");
    }
    return 0;
}

4.8 각 정점의 차수 구하기

#include <stdio.h>

int main() {
    int n, m;
    int graph[101][101] = {0};

    scanf("%d %d", &n, &m);

    for (int i = 0; i < m; i++) {
        int a, b;
        scanf("%d %d", &a, &b);
        graph[a][b] = 1;
        graph[b][a] = 1;
    }

    for (int i = 1; i <= n; i++) {
        int degree = 0;
        for (int j = 1; j <= n; j++) {
            if (graph[i][j] == 1) degree++;
        }
        printf("%d번 정점 차수: %d\n", i, degree);
    }
    return 0;
}

5. 차수와 연결 질문

5.1 무방향 그래프 차수

한 행에서 1의 개수를 세면 정점의 차수가 됩니다.

degree = graph[i][1] + graph[i][2] + ... + graph[i][n]

5.2 방향 그래프 진출/진입차수

진출차수는 행, 진입차수는 열을 봅니다.

out = graph[i][j], in = graph[j][i]

5.3 연결 여부 확인

인접 행렬은 두 정점의 연결 여부를 바로 확인할 수 있습니다.

graph[a][b] == 1

6. 플로이드 워셜용 거리 행렬

최단거리 문제에서는 연결되지 않은 곳을 0으로 두면 안 됩니다. 0은 거리 0이라는 의미가 될 수 있기 때문입니다.

#define INF 1000000000

#include <stdio.h>

#define INF 1000000000

int main() {
    int n, m;
    int dist[101][101];

    scanf("%d %d", &n, &m);

    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= n; j++) {
            if (i == j) dist[i][j] = 0;
            else dist[i][j] = INF;
        }
    }

    for (int i = 0; i < m; i++) {
        int a, b, w;
        scanf("%d %d %d", &a, &b, &w);
        dist[a][b] = w;
    }

    return 0;
}
